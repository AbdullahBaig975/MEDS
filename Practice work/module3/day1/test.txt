text

.globl main

main:

addi a0, zero, 5    # a0 = 5
addi a1, zero, 10   # a1 = 10
add a2, a0, a1      # a2 = 15

# Print result
addi a0, zero, 1    # ecall 1 = print int
mv a1, a2           # a1 = value to print

ecall               # Exit
addi a0, zero, 10   # ecall 10 = exit
