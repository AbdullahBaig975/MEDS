.text
.globl main

main:
    li a0, 10          # Calculate fib(10)
    jal ra, fib
    
    # Print result
    li a7, 1           # Expected: 55
    ecall
    
    # Exit
    li a7, 10
    ecall

# --- Recursive fib(a0) -> a0 ---
fib:
    # Prologue: allocate stack and preserve registers
    addi sp, sp, -16
    sw ra, 12(sp)
    sw s0, 8(sp)
    sw s1, 4(sp)

    mv s0, a0          # s0 = n

    # Base Cases: n <= 0 -> 0; n == 1 -> 1
    li t0, 0
    bgt s0, t0, check_one
    li a0, 0           # return 0
    j fib_return

check_one:
    li t0, 1
    bne s0, t0, recurse
    li a0, 1           # return 1
    j fib_return

recurse:
    # Compute fib(n - 1)
    addi a0, s0, -1    # arg = n - 1
    jal ra, fib
    mv s1, a0          # s1 = fib(n - 1)

    # Compute fib(n - 2)
    addi a0, s0, -2    # arg = n - 2
    jal ra, fib        # a0 = fib(n - 2)

    # Sum both recursive branches
    add a0, s1, a0     # a0 = fib(n - 1) + fib(n - 2)

fib_return:
    # Epilogue: restore stack and registers
    lw ra, 12(sp)
    lw s0, 8(sp)
    lw s1, 4(sp)
    addi sp, sp, 16
    jr ra
