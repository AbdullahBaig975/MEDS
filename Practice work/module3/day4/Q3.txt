.text
.globl main

main:
    li a0, 10          # Initialize test value
    jal ra, parent
    
    # Exit
    li a7, 10
    ecall

# --- Function: parent(a0) ---
parent:
    # Prologue: Store ra and s0 on stack
    addi sp, sp, -16
    sw ra, 12(sp)
    sw s0, 8(sp)

    mv s0, a0          # Store input argument in saved register s0
    
    addi a0, s0, 5     # Modify argument for child
    jal ra, child      # Call child function (overwrites ra)
    
    # After returning, s0 is guaranteed to still hold our original 10
    add a0, a0, s0     # final result = child(15) + s0 = 30 + 10 = 40

    # Epilogue: Restore original ra and s0
    lw ra, 12(sp)
    lw s0, 8(sp)
    addi sp, sp, 16
    jr ra

# --- Function: child(a0) ---
child:
    # Leaf function: doesn't call others, so no stack preservation needed
    slli a0, a0, 1     # Double the input (15 * 2 = 30)
    jr ra
